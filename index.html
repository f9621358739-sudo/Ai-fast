import os import time import urllib.parse import webbrowser import speech_recognition as recognizer_lib from gtts import gTTS import pyttsx3 import pyautogui from google import genai # ------------------------------------------------------------- # 1. अपेक्स (Apex) क्लाइंट सेटअप (अपनी API Key यहाँ डालें) # ------------------------------------------------------------- os.environ["GEMINI_API_KEY"] = "YOUR_API_KEY" # यहाँ अपनी असली API Key डालें client = genai.Client() def speak_text(text): print(f"\n🔊 अपेक्स का जवाब: {text}") try: tts = gTTS(text=text, lang='hi', slow=False) tts.save("response.mp3") if os.name == 'nt': os.system("start response.mp3") else: os.system("afplay response.mp3" if os.uname().sysname == "Darwin" else "mpg321 response.mp3") except Exception as e: engine = pyttsx3.init() engine.say(text) engine.runAndWait() # आवाज़ सुनने के लिए def listen_to_user(): r = recognizer_lib.Recognizer() with recognizer_lib.Microphone() as source: print("\n🎤 अपेक्स सुन रहा है... (अपनी कमांड बोलिए)") r.adjust_for_ambient_noise(source, duration=0.5) try: audio = r.listen(source, timeout=5, phrase_time_limit=15) text = r.recognize_google(audio, language="hi-IN") print(f"👤 आपने कहा: {text}") return text.lower() except Exception as e: return None # बैकग्राउंड में 'Hello Apex' सुनने के लिए def wait_for_wake_word(): r = recognizer_lib.Recognizer() with recognizer_lib.Microphone() as source: r.adjust_for_ambient_noise(source, duration=0.8) while True: print("💤 (अपेक्स बैकग्राउंड में वेक वर्ड 'Hello Apex' का इंतज़ार कर रहा है...)") try: audio = r.listen(source) text = r.recognize_google(audio, language="hi-IN").lower() print(f"सुना गया: {text}") if "हैलो अपेक्स" in text or "hello apex" in text or "apex" in text or "अपेक्स" in text: print("⚡ वेक वर्ड डिटेक्ट हो गया!") return True except Exception: continue # ------------------------------------------------------------- # 2. एडवांस ऐप्स, म्यूजिक और ऑटोमेशन इंजन # ------------------------------------------------------------- def process_advanced_commands(command): # A. डिस्क़ो डांस ट्रैक या कोई भी गाना/म्यूजिक YouTube Music / YouTube पर बजाना if "disco" in command or "डिस्को" in command or "dance track" in command or "डांस ट्रैक" in command or "गाना" in command or "song" in command: search_query = command for word in ["youtube", "यूट्यूब", "पर", "सर्च", "करके", "बजा दो", "मंगा दो", "चला दो", "सॉन्ग", "गाना", "करो", "music", "म्यूजिक"]: search_query = search_query.replace(word, "") search_query = search_query.strip() # यदि यूजर ने सिर्फ 'डिस्को डांस' कहा है, तो बढ़िया सा ट्रैक सेट कर दें if not search_query or "disco" in search_query or "डिस्को" in search_query: search_query = "best disco dance track 2026 club mix" speak_text(f"यूट्यूब म्यूजिक से {search_query} चला रहा हूँ। कमर की पेटी बांध लीजिए!") encoded_query = urllib.parse.quote(search_query) # YouTube Music लिंक ओपन करना webbrowser.open(f"https://music.youtube.com/search?q={encoded_query}") time.sleep(3) pyautogui.press('tab') pyautogui.press('enter') return True # B. यूट्यूब पर वीडियो/सर्च करना elif "youtube" in command or "यूट्यूब" in command: search_query = command for word in ["youtube", "यूट्यूब", "पर", "सर्च", "करके", "बजा दो", "मंगा दो", "चला दो", "सॉन्ग", "गाना", "करो"]: search_query = search_query.replace(word, "") search_query = search_query.strip() if search_query: speak_text(f"यूट्यूब पर {search_query} सर्च कर रहा हूँ।") encoded_query = urllib.parse.quote(search_query) webbrowser.open(f"https://www.youtube.com/results?search_query={encoded_query}") time.sleep(3) pyautogui.press('tab') pyautogui.press('enter') else: speak_text("यूट्यूब खोल रहा हूँ।") webbrowser.open("https://www.youtube.com") return True # C. व्हाट्सएप पर मैसेज भेजना elif "whatsapp" in command or "व्हाट्सएप" in command: speak_text("व्हाट्सएप वेब खोल रहा हूँ...") webbrowser.open("https://web.whatsapp.com") time.sleep(8) speak_text("किसे मैसेज भेजना है और क्या लिखना है? कृपया बोलें (जैसे: राजन को लिखो हे आई एम आशीष)") msg_command = listen_to_user() if msg_command: speak_text("मैसेज टाइप करके सेंड कर रहा हूँ...") pyautogui.hotkey('ctrl', 'alt', '/') time.sleep(1) pyautogui.write(msg_command, interval=0.05) pyautogui.press('enter') time.sleep(1) pyautogui.press('enter') return True # D. गूगल या वेब पर कुछ भी सर्च करना elif "सर्च करो" in command or "search" in command or "गूगल" in command or "google" in command: search_query = command for word in ["google", "गूगल", "पर", "सर्च", "करो", "ढूंढो"]: search_query = search_query.replace(word, "") search_query = search_query.strip() if search_query: speak_text(f"गूगल पर {search_query} सर्च कर रहा हूँ।") webbrowser.open(f"https://www.google.com/search?q={urllib.parse.quote(search_query)}") return True # E. नोटपैड खोलकर लिखना elif "नोटपैड" in command or "notepad" in command: speak_text("नोटपैड खोल रहा हूँ। आप जो बोलेंगे, मैं लिख दूंगा।") os.system("notepad") time.sleep(2) speak_text("बताइए नोटपैड में क्या लिखना है?") text_to_write = listen_to_user() if text_to_write: pyautogui.write(text_to_write, interval=0.05) speak_text("नोटपैड में लिख दिया गया है।") return True # F. स्क्रीनशॉट लेना elif "स्क्रीनशॉट" in command or "screenshot" in command: speak_text("स्क्रीनशॉट ले रहा हूँ।") screenshot = pyautogui.screenshot() screenshot.save("apex_screenshot.png") speak_text("स्क्रीनशott सेव हो गया है।") return True # G. करंट विंडो बंद करना elif "विंडो बंद करो" in command or "close" in command: speak_text("विंडो बंद कर रहा हूँ।") pyautogui.hotkey('alt', 'f4') return True return False # ------------------------------------------------------------- # 3. मुख्य प्रोग्राम लूप (सटीक जवाब और वेक वर्ड इंजन) # ------------------------------------------------------------- if __name__ == "__main__": print("=== 'Apex' Ultimate AI Assistant एक्टिव हो गया है ===") print("💡 टिप: बैकग्राउंड में 'Hello Apex' बोलें, और अपने हर सवाल का सटीक जवाब पाएं!") while True: if wait_for_wake_word(): speak_text("हाँ, बोलिए मैं सुन रहा हूँ!") user_input = listen_to_user() if user_input: if "बाहर निकलो" in user_input or "exit" in user_input or "बंद हो जाओ" in user_input: speak_text("अलविदा! फिर मिलेंगे।") break # 1. पहले चेक करेगा कि क्या कोई ऐप, व्हाट्सएप या गाना (डिस्को ट्रैक) बजाना है is_action_done = process_advanced_commands(user_input) # 2. अगर कोई ऐप कमांड नहीं है, तो जेमिनी मॉडल से बिल्कुल सटीक और सही जवाब प्राप्त करेगा if not is_action_done: print("🤖 अपेक्स सोच रहा है (सटीक जवाब ढूंढ रहा है)...") try: # सुनिश्चित करने के लिए कि जवाब एकदम सही और स्पष्ट हो response = client.models.generate_content( model='gemini-2.5-flash', contents=f"कृपया इस सवाल का एकदम सटीक, सही और स्पष्ट जवाब दें: {user_input}", ) ai_reply = response.text speak_text(ai_reply) except Exception as e: speak_text(f"क्षमा करें, जवाब ढूंढने में त्रुटि आई है: {e}")
